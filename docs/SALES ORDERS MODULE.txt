SCOPE:
Build the complete Sales Orders feature exactly as APPENDIX A defines it:
(A) Register: backlog KPI strip, status filter tiles, search/sort, delivery-intelligence chips,
    per-order fulfilment progress bars, Views / Due-for-delivery / Documents rail,
    row-wide click-to-view, status-aware row actions (number-only Order № column).
(B) Create/Edit Sales Order page: existing vs walk-in customer, delivery scheduling,
    warehouse/cost-centre, line editor with on-hand stock checks and completeness ✓,
    live summary rail with "Release check" gate, Save Draft / Release Order actions.
(C) View modal: Draft→Sent→Confirmed→Fulfilled timeline, ordered-vs-fulfilled lines,
    delivery & lineage chips, status-aware footer (print/send/confirm/fulfil/invoice/…).
(D) Fulfilment modal: partial deliveries with per-line remaining caps, delivery-note №,
    received-by; completion flips order to Fulfilled + success confirmation.
(E) Print preview: A4 Sales Order/Acknowledgement sheet AND price-free Delivery Note sheet,
    each with Print + Save as PDF (server PDF endpoint live).
(F) Email dispatch of the order acknowledgement to the customer (format-validated, audited).
(G) One-click invoice conversion from fulfilled orders with cross-links + success box.

HARD GUARD — do NOT modify unrelated modules (quotations, invoicing, stock masters) beyond
the sanctioned integration points in §9. Frontend view-layer rebuild + §9 contracts only.

     live "Release check" row: green "Ready ✓" or amber "Missing: …" listing each unmet
     condition; the Release button title mirrors it. Complete lines carry a green ✓ in the
     Amount cell; summary note counts them.
R10 · WALK-IN RULE: unknown customer name creates the customer record in the SAME DB
     transaction and links it; registry immediately feeds searches/datalists; toast cites
     the new code. Existing-mode match locks contact/email/phone/address (grey readonly).

  /* ---------- lifecycle ---------- */
  function act(a, id) {
    var tr = rowOf(id); if (!tr) return;
    var ref = tr.querySelector('.ono').textContent;
    if (a === 'view') { openView(id); return; }
    if (a === 'edit') {
      if (tr.dataset.status !== 'draft') { toast('🔒 ' + ref + ' is ' + tr.dataset.status + ' — only drafts can be edited'); return; }
      $('viewScrim').classList.remove('on');
      openEditOrder(id);
      return;
    }
    if (a === 'fulfil') { openFulfil(id); return; }
    if (a === 'invoice') { createInvoice(id); return; }
    if (a === 'send') { sendOrder(id); return; }
    if (a === 'printOrder') { openOrderPrint(id, 'order'); return; }
    if (a === 'printDN') { openOrderPrint(id, 'delivery'); return; }
    if (a === 'dup') { toast('⧉ ' + ref + ' duplicated as new draft'); return; }
    if (a === 'release') ask({ tone: 'primary', title: 'Release ' + ref + '?', msg: 'Sends the order to the warehouse queue and emails the acknowledgement to the customer. Status moves to <b>Sent</b>.', okLabel: 'Release now', onOk: function () { setStatus(id, 'sent'); toast('📨 ' + ref + ' released'); } });
    if (a === 'confirm') ask({ tone: 'ok', title: 'Confirm ' + ref + '?', msg: 'Allocates stock from the order warehouse and locks pricing. The order becomes <b>Confirmed</b> and eligible for fulfilment.', okLabel: 'Confirm order', onOk: function () { setStatus(id, 'confirmed'); toast('✓ ' + ref + ' confirmed — stock allocated'); } });
    if (a === 'cancel') ask({ tone: 'danger', title: 'Cancel ' + ref + '?', msg: 'Cancels the order and releases any allocated stock. Cancelled orders remain for reporting and can be duplicated.', okLabel: 'Cancel order', onOk: function () { setStatus(id, 'cancelled'); toast('✕ ' + ref + ' cancelled'); } });
    if (a === 'void') ask({ tone: 'danger', title: 'Void ' + ref + '?', msg: 'Voids the draft order permanently. Void documents are retained for audit only.', okLabel: 'Void order', onOk: function () { setStatus(id, 'void'); toast('⃠ ' + ref + ' voided'); } });
  }

  function lineValid(r, amt) {
    return r.querySelector('[data-f="code"]').value.trim() !== '' &&
      r.querySelector('[data-f="name"]').value.trim() !== '' &&
      num(r.querySelector('[data-f="q"]').value) > 0 &&
      num(r.querySelector('[data-f="p"]').value) > 0 &&
      amt > 0;
  }
  function coRecompute() {
    var sub = 0, n = 0;
    Array.prototype.forEach.call($('coLines').rows, function (r) {
      var q = num(r.querySelector('[data-f="q"]').value), p = num(r.querySelector('[data-f="p"]').value);
      var amt = q * p; sub += amt;
      var ok = lineValid(r, amt); if (ok) n++;
      r.querySelector('.ramt').innerHTML = fmt(amt) + (ok ? '<span class="lok">✓</span>' : '');
      var code = r.querySelector('[data-f="code"]').value.trim();
      var oh = code && ITEMS.hasOwnProperty(code) ? ITEMS[code] : null;
      var cellEl = r.querySelector('.ohcell');
      cellEl.textContent = oh == null ? '—' : (oh != null && q > oh ? oh + ' ⚠' : oh);
      cellEl.classList.toggle('over', oh != null && q > oh);
    });
    lineCount = n;
    var tax = sub * 0.165;
    $('soSub').textContent = fmt(sub); $('soTax').textContent = fmt(tax); $('soGrand').textContent = 'K ' + fmt(sub + tax);
    $('soLines').textContent = n ? n + ' complete line' + (n === 1 ? '' : 's') + ' (code · name · qty · price)' : 'No complete lines yet';
    updateBtns();
    return { sub: sub, tax: tax, total: sub + tax, n: n };
  }

  function collectLines() {
    var lines = [], sub = 0;
    Array.prototype.forEach.call($('coLines').rows, function (r) {
      var code = r.querySelector('[data-f="code"]').value.trim();
      var nm = r.querySelector('[data-f="name"]').value.trim();
      var ds = r.querySelector('[data-f="desc"]').value.trim();
      var q = num(r.querySelector('[data-f="q"]').value), p = num(r.querySelector('[data-f="p"]').value);
      if (code && nm && q > 0 && p > 0) { lines.push([code + ' — ' + nm + (ds ? ' (' + ds + ')' : ''), q, 0, fmt(p), fmt(q * p)]); sub += q * p; }
    });
    return { lines: lines, sub: sub };
  }
  function saveOrder(release) {
    var walkin = $('coMode').value === 'walkin';
    var name = walkin ? $('coName').value.trim() : $('coSearch').value.trim();
    if (!name) { (walkin ? $('coName') : $('coSearch')).classList.add('bad'); toast(walkin ? 'Customer name is required' : 'Select a customer from the registry'); return false; }
    (walkin ? $('coName') : $('coSearch')).classList.remove('bad');
    if (!walkin && !matchedCode) { $('coSearch').classList.add('bad'); toast('Pick a matching customer from the list'); return false; }
    $('coSearch').classList.remove('bad');
    var c = collectLines();
    if (!c.lines.length) { toast('Add at least one complete line (code, item name, qty, unit price)'); return false; }
    if (release && !$('coDelivery').value) { $('coDelivery').classList.add('bad'); toast('Delivery date is required to release'); return false; }
    $('coDelivery').classList.remove('bad');
    var email = $('coEmail').value.trim();
    var savedCode = '';
    if (walkin) { savedCode = 'CUS-' + String(custSeq++).padStart(4, '0'); CUSTOMERS.push({ code: savedCode, name: name, email: email, phone: $('coPhone').value.trim(), address: $('coAddress').value.trim() }); fillCustDatalist(); }
    else savedCode = matchedCode;
    var tax = c.sub * 0.165, total = c.sub + tax;
    var st = release ? 'sent' : 'draft';
    var ref, id;
    if (editId) {
      id = editId; ref = rowOf(id).querySelector('.ono').textContent;
      var od = OD[id];
      od.cust = name; od.contact = $('coContact').value.trim() || name; od.email = email; od.phone = $('coPhone').value.trim(); od.address = $('coAddress').value.trim();
      od.method = $('coMethod').value; od.wh = $('coWh').value; od.cc = $('coCC').value; od.terms = TERMS[$('coTerms').value];
      od.lines = c.lines; od.sub = c.sub; od.tax = tax; od.notes = $('coNotesC').value.trim(); od.internal = $('coNotesI').value.trim();
      var p = rowPayload(id, st);
      p.total = total; p.delivery = $('coDelivery').value || p.delivery; p.date = $('coDate').value; p.name = name; p.search = (name + ' ' + ref).toLowerCase();
      replaceRow(id, p);
    } else {
      ref = 'SO-2026-' + String(nextOrder++).padStart(4, '0');
      id = 'o' + (rowSeq++);
      OD[id] = { cust: name, contact: $('coContact').value.trim() || name, email: email, phone: $('coPhone').value.trim(), address: $('coAddress').value.trim() || '—', method: $('coMethod').value, wh: $('coWh').value, cc: $('coCC').value, terms: TERMS[$('coTerms').value], source: $('coRef').value.trim().match(/^QT-/) ? $('coRef').value.trim() : '', invoice: '', lines: c.lines, sub: c.sub, tax: tax, notes: $('coNotesC').value.trim(), internal: $('coNotesI').value.trim() };
      insertRow({ id: id, ref: ref, name: name, date: $('coDate').value, delivery: $('coDelivery').value || addDaysISO(TODAY_ISO, 7), total: total, status: st, search: (name + ' ' + ref + (OD[id].source ? ' ' + OD[id].source : '')).toLowerCase() });
      if (release && emailOk(email)) { OD[id].emailed = email; OD[id].emailedAt = TODAY_ISO; }
    }
    editId = null; matchedCode = null;
    $('coSearch').value = ''; $('coName').value = ''; $('coContact').value = ''; $('coEmail').value = ''; $('coPhone').value = ''; $('coAddress').value = '';
    $('coRef').value = ''; $('coNotesC').value = ''; $('coNotesI').value = '';
    $('coMode').value = 'existing'; $('coMode').dispatchEvent(new Event('change'));
    $('coLines').innerHTML = coLineRow(); $('coDate').value = TODAY_ISO; $('coDelivery').value = addDaysISO(TODAY_ISO, 7);
    setLock(false); coRecompute(); coSummary();
    showView('list'); refresh();
    toast('✓ ' + ref + ' ' + (release ? 'released to warehouse & customer' : 'saved as draft') + ' · ' + name + (walkin ? ' · new customer saved (' + savedCode + ')' : ''));
    return true;
  }
