  var toastEl = document.getElementById('toast'), toastT;
  function toast(m) { toastEl.textContent = m; toastEl.classList.add('on'); clearTimeout(toastT); toastT = setTimeout(function () { toastEl.classList.remove('on'); }, 3000); }

  var pview = document.getElementById('printView');
  function openPreview() {
    pview.classList.add('on');
    pview.setAttribute('aria-hidden', 'false');
    document.body.classList.add('pv-on');
    pview.scrollTop = 0;
    toast('Print preview ready — use Print (or Ctrl+P) in the toolbar');
  }
  function closePreview() {
    pview.classList.remove('on');
    pview.setAttribute('aria-hidden', 'true');
    document.body.classList.remove('pv-on');
  }

  function safePrint() {
    document.body.classList.add('pv-on');
    pview.classList.add('on');
    if (typeof window.print !== 'function') { fallbackPrint(); return; }
    var ok = true;
    try { window.print(); } catch (e) { ok = false; }
    if (!ok) fallbackPrint();
  }

  function fallbackPrint() {
    var css = document.getElementById('appStyle').innerHTML;
    var sheet = document.getElementById('voucherSheet').outerHTML;
    var w = window.open('', '_blank');
    if (!w) { toast('Pop-ups blocked — press Ctrl+P while the preview is open'); return; }
    w.document.open();
    w.document.write(
      '<!DOCTYPE html><html><head><title>JE-2026-0018 · Journal Voucher</title>' +
      '<style>' + css + '</style>' +
      '<style>@page{size:A4;margin:10mm}*{-webkit-print-color-adjust:exact;print-color-adjust:exact}' +
      'body{background:#fff!important;overflow:visible!important;padding:0!important}' +
      '.toolbar,.toast{display:none!important}' +
      '.pview{position:static!important;display:block!important;overflow:visible!important;background:#fff!important}' +
      '.sheet-wrap{padding:0!important;display:block!important}' +

      '</head><body class="pv-on"><div class="pview on"><div class="sheet-wrap">' + sheet + '</div></div></body></html>'
    );
    w.document.close();
    w.focus();
    setTimeout(function () { try { w.print(); } catch (e) {} }, 400);
  }

  document.getElementById('btnPrint').addEventListener('click', openPreview);
  document.getElementById('pvBack').addEventListener('click', closePreview);
  document.getElementById('pvPrint').addEventListener('click', safePrint);

  document.addEventListener('keydown', function (e) {
    if ((e.ctrlKey || e.metaKey) && (e.key === 'p' || e.key === 'P')) {

    }
    if (e.key === 'Escape') {
      if (document.getElementById('revScrim').classList.contains('on')) closeRev();
      else if (pview.classList.contains('on')) closePreview();
    }
  });
