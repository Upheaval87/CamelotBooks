==================== 6 · SANCTIONED BACKEND CONTRACTS ====================
6.1 Schema (add missing columns via migration, nullable-safe):

    opening_balance_date, opening_journal_id (nullable FK), notes, status, created_at ("since").

    opening_balance ≠ 0 posts a journal dated opening_balance_date, source
    "customer_opening", lines: Dr Accounts Receivable (customer) / Cr Opening Balance Equity
    (reversed when negative); store opening_journal_id; cross-link shown on the journal.
    Editing opening balance after transactions exist is BLOCKED server-side (toast explains
    the adjustment-journal route).
6.3 Credit limit: stored per customer + currency; exposed to invoice creation as a
    warning/block flag per existing settings — do not change invoice flow here.
6.4 Audit: create/update/delete/opening-journal events logged with user + timestamp;
    delete is a hard delete only under the §5.2 guard, else inactive-status is the remedy.
6.5 Terms keys: receipt|net15|net30|net60|custom with days integer; due-date calculation on
    documents uses days (preset map or custom value).

    $('fTerms2').classList.remove('bad');
    var days = Number($('fDays').value || 0);
    if (terms === 'custom' && !(days > 0)) { $('fDays').classList.add('bad'); toast('Custom terms need a day count'); return; }
    $('fDays').classList.remove('bad');
    var opening = Number($('fOpen').value || 0);
    var openDate = $('fOpenDate').value;
    if (opening !== 0 && !openDate) { $('fOpenDate').classList.add('bad'); toast('Opening balance date is required'); return; }
    $('fOpenDate').classList.remove('bad');

    var data = {
      name: name,
      display: $('fDisplay').value.trim(),
      email: $('fEmail').value.trim(),
      phone: $('fPhone').value.trim(),
      bill: $('fBill').value.trim(),
      ship: $('fShip').value.trim(),
      cur: $('fCur').value,
      terms: terms,
      days: terms === 'custom' ? days : ({ receipt: 0, net15: 15, net30: 30, net60: 60 })[terms],
      limit: Number($('fLimit').value || 0),
      opening: opening,
      openDate: openDate
    };

    if (editingId) {
      var c = CUST[editingId];
      Object.assign(c, data);
      c.balance = opening;
      var tr = document.querySelector('#custBody tr[data-id="' + editingId + '"]');
      if (tr) tr.outerHTML = rowHTML(c);
      close('formScrim'); counts(); apply();
      toast('✓ ' + c.name + ' updated');
    } else {
      var id = 'c' + (seq++);
      CUST[id] = Object.assign({ id: id, code: 'CUS-000' + (Object.keys(CUST).length + 1), since: 'Oct 2026', status: 'active', balance: opening, open: 0, sales: 0, act: [] }, data);
      var tmp = document.createElement('tbody');
      tmp.innerHTML = rowHTML(CUST[id]);
      $('custBody').insertBefore(tmp.firstChild, $('noRes'));
      close('formScrim'); counts(); apply();
      toast('✓ ' + CUST[id].name + ' created — ' + CUST[id].code);
    }
  });

  /* ---------- delete ---------- */
  function openDel(id) {
    var c = CUST[id]; if (!c) return;
    deletingId = id;
    $('delTitle').textContent = 'Delete ' + c.name + '?';
    $('delMsg').innerHTML = 'This permanently removes <b>' + c.code + '</b> and its contact record. Only customers with a zero balance and no open documents can be deleted; posted history is retained for audit.';
    open('delScrim');
  }
  $('delOk').addEventListener('click', function () {
    if (!deletingId) return;
    var c = CUST[deletingId];
    var tr = document.querySelector('#custBody tr[data-id="' + deletingId + '"]');
    if (tr) tr.remove();
    delete CUST[deletingId];
    deletingId = null;
    close('delScrim'); counts(); apply();
    toast('🗑 ' + (c ? c.name : 'Customer') + ' deleted');
  });

  /* ---------- row delegation ---------- */
  $('custBody').addEventListener('click', function (e) {
    var b = e.target.closest('button.ib'); if (!b) return;
    var id = b.closest('tr').dataset.id;
    if (b.dataset.act === 'view') openView(id);
    if (b.dataset.act === 'edit') openForm(id);
    if (b.dataset.act === 'delete') openDel(id);
  });

  counts();
  apply();
})();
