    requote creation; converted sets converted_invoice_id.

    rate); words-amount rendered server-side for PDF parity.
9.6 Audit: every transition + customer auto-create logged (user, ts, from→to, reason
    where captured); deletes not offered — void is the remedy.

      '<td><input class="lin" data-f="name" value="' + String(name || '').replace(/"/g, '&quot;') + '" placeholder="Search / type item…"></td>' +
      '<td><input class="lin" data-f="desc" value="' + String(desc || '').replace(/"/g, '&quot;') + '" placeholder="Description"></td>' +
      '<td><input class="lin num" data-f="q" type="number" min="0" step="1" value="' + (q == null ? 1 : q) + '"></td>' +
      '<td><input class="lin num" data-f="p" type="number" min="0" step="0.01" value="' + (p || 0) + '"></td>' +
      '<td><input class="lin num" data-f="d" type="number" min="0" max="100" step="0.5" value="' + (d || 0) + '"></td>' +
      '<td class="amtcell ramt">0.00</td>' +
      '<td style="white-space:nowrap"><button class="xb dup" type="button" title="Duplicate line">' + SVG_DUP + '</button> <button class="xb" type="button" title="Remove line">' + SVG_X + '</button></td></tr>';
  }
  function ciRecompute() {
    var sub = 0, n = 0;
    Array.prototype.forEach.call($('ciLines').rows, function (r) {
      var q = num(r.querySelector('[data-f="q"]').value), p = num(r.querySelector('[data-f="p"]').value), d = num(r.querySelector('[data-f="d"]').value);
      var amt = q * p * (1 - d / 100); sub += amt;
      r.querySelector('.ramt').textContent = fmt(amt);
      if (r.querySelector('[data-f="name"]').value.trim()) n++;
    });
    var tax = sub * 0.165;
    $('sumSub').textContent = fmt(sub); $('sumTax').textContent = fmt(tax);
    $('sumGrand').textContent = curSym + ' ' + fmt(sub + tax);
    $('sumLines').textContent = n ? n + ' line item' + (n === 1 ? '' : 's') + ' · tax 16.5%' : 'No line items yet';
    return { sub: sub, tax: tax, total: sub + tax, n: n };
  }
  function ciSummaryMeta() {
    var walkin = $('ciMode').value === 'walkin';
    $('sumCust').textContent = walkin ? ($('ciName').value.trim() || '—') : ($('ciSearch').value.trim() || '—');

    $('sumValid').textContent = dfmt($('ciValid').value);
  }

  function saveQuotation(finalize, print) {
    var walkin = $('ciMode').value === 'walkin';
    var name = walkin ? $('ciName').value.trim() : $('ciSearch').value.trim();
    if (!name) { (walkin ? $('ciName') : $('ciSearch')).classList.add('bad'); toast(walkin ? 'Customer name is required' : 'Select a customer from the registry'); return false; }
    (walkin ? $('ciName') : $('ciSearch')).classList.remove('bad');
    if (!walkin && !matchedCode) { $('ciSearch').classList.add('bad'); toast('Pick a matching customer from the list'); return false; }
    $('ciSearch').classList.remove('bad');
    if (!$('ciValid').value) { $('ciValid').classList.add('bad'); toast('Valid-until date is required'); return false; }
    $('ciValid').classList.remove('bad');
    if ($('ciValid').value < $('ciDate').value) { $('ciValid').classList.add('bad'); toast('Valid-until cannot be before the quotation date'); return false; }
    $('ciValid').classList.remove('bad');
    var t = ciRecompute();
    if (!t.n) { toast('Add at least one line item with a name and quantity'); return false; }
    var lines = [], sub = 0;
    Array.prototype.forEach.call($('ciLines').rows, function (r) {
      var nm = r.querySelector('[data-f="name"]').value.trim();
      var q = num(r.querySelector('[data-f="q"]').value), p = num(r.querySelector('[data-f="p"]').value), d = num(r.querySelector('[data-f="d"]').value);
      if (nm && q > 0) {
        var code = r.querySelector('[data-f="code"]').value.trim();
        var desc = r.querySelector('[data-f="desc"]').value.trim();
        lines.push([(code ? code + ' — ' : '') + nm + (desc ? ' (' + desc + ')' : ''), q, fmt(p), fmt(q * p * (1 - d / 100))]);
        sub += q * p * (1 - d / 100);
      }
    });
    var tax = sub * 0.165, total = sub + tax;
    var ref = 'QT-2026-' + String(nextNum).padStart(4, '0');
    var savedCode = '';
    if (walkin) { savedCode = 'CUS-' + String(custSeq++).padStart(4, '0'); CUSTOMERS.push({ code: savedCode, name: name, email: $('ciEmail').value.trim(), phone: $('ciPhone').value.trim(), terms: $('ciTerms').value }); fillCustDatalist(); }
    else savedCode = matchedCode;
    var newId = 'q' + (rowSeq++);
    QD[newId] = {
      sub: sub, disc: 0, tax: tax, cur: curSym, lines: lines,
      notes: $('ciNotesC').value.trim() || '—', terms: TERMS[$('ciTerms').value],
      email: $('ciEmail').value.trim(), phone: $('ciPhone').value.trim(), cust: name,
      contact: $('ciContact').value.trim() || name, internal: $('ciNotesI').value.trim(),
      branch: $('ciBranch').value, cc: $('ciCC').value, prepared: 'Elvis Seyama', refNo: $('ciRef').value.trim(), project: $('ciProject').value.trim()
    };
    var st = finalize ? 'sent' : 'draft';
    insertQuoteRow({ id: newId, ref: ref, sub: ($('ciRef').value.trim() ? 'Ref ' + $('ciRef').value.trim() : 'Ref —') + ' · ' + (finalize ? 'issued' : 'draft'), name: name, date: $('ciDate').value, valid: $('ciValid').value, total: total, status: st, search: (name + ' ' + ref).toLowerCase() });
    nextNum++;
    matchedCode = null;
    $('ciSearch').value = ''; $('ciName').value = ''; $('ciContact').value = ''; $('ciEmail').value = ''; $('ciPhone').value = '';
    $('ciTerms').value = 'net30'; $('ciRef').value = ''; $('ciProject').value = ''; $('ciNotesC').value = ''; $('ciNotesI').value = '';
    $('ciMode').value = 'existing'; $('ciMode').dispatchEvent(new Event('change'));
    $('ciLines').innerHTML = ciLineRow(); $('ciDate').value = TODAY_ISO; $('ciValid').value = addDays(TODAY_ISO, 14);
    setExistingLock(false); ciRecompute(); ciSummaryMeta();
    showView('list'); refresh();
    if (finalize && print) openQuotePrint(newId);
    toast('✓ ' + ref + ' ' + (finalize ? (print ? 'finalized & issued — printable quotation opened' : 'finalized & issued') : 'saved to database — send later') + ' · ' + name + (walkin ? ' · new customer saved (' + savedCode + ')' : ''));
    return true;
  }

  function act(actName, id) {
    var tr = rowOf(id); if (!tr) return;
    var ref = tr.querySelector('.qno').textContent;
    if (actName === 'view') { openView(id); return; }
    if (actName === 'requote') { openRequote(id); return; }
    if (actName === 'edit') { openEdit(id); return; }
    if (actName === 'print') { openQuotePrint(id); return; }
    if (actName === 'dup') { toast('⧉ ' + ref + ' duplicated as new draft'); return; }
    if (actName === 'send') ask({ tone: 'primary', title: 'Send ' + ref + ' to customer?', msg: 'The quotation PDF is emailed to the customer and the status moves to <b>Sent</b>. The validity clock starts today.', okLabel: 'Send now', onOk: function () { setStatus(id, 'sent'); toast('📨 ' + ref + ' sent to customer'); } });
    if (actName === 'accept') ask({ tone: 'ok', title: 'Mark ' + ref + ' as accepted?', msg: 'Records the customer acceptance. You can then convert the quotation into a sales invoice.', okLabel: 'Mark accepted', onOk: function () { setStatus(id, 'accepted'); toast('✓ ' + ref + ' accepted'); } });
    if (actName === 'decline') ask({ tone: 'danger', title: 'Mark ' + ref + ' as declined?', msg: 'Records the lost deal for pipeline reporting. The quotation can still be re-quoted or duplicated later.', okLabel: 'Mark declined', onOk: function () { setStatus(id, 'declined'); toast('✕ ' + ref + ' marked declined'); } });
    if (actName === 'convert') ask({ tone: 'primary', title: 'Convert ' + ref + ' to invoice?', msg: 'Creates a sales invoice from the accepted quotation lines, locks the quotation and cross-links both documents.', okLabel: 'Convert now', onOk: function () { setStatus(id, 'converted'); toast('⇄ ' + ref + ' converted — invoice created'); } });
    if (actName === 'void') ask({ tone: 'danger', title: 'Void ' + ref + '?', msg: 'Voids the quotation permanently. Void documents are retained for audit and excluded from pipeline metrics.', okLabel: 'Void quotation', onOk: function () { setStatus(id, 'void'); toast('⃠ ' + ref + ' voided'); } });
  }
