Build the complete Invoices (AR) feature exactly as APPENDIX A defines it:
(A) Register: AR KPI strip, status tiles incl. derived Overdue, search/sort, toggleable
    date-period filter, aging chips, view-only Actions column, row-wide click-to-workspace.

    ("View all →") with AR/overdue/due tiles and aging-sorted open-invoice table.

    Post Invoice with confirmation, sticky Summary + Quick-nav rail.

    (Overview / Line items / Payments), status-aware footer actions, trailing Close.

    numbering, balance-after preview, itemised CONFIRMATION before posting, immutable
    receipts, settlement success box.

(G) Print preview: A4 Tax Invoice opened in a NEW TAB (Print / Save as PDF / Close tab),
    in-app overlay only as popup-blocked fallback.

  function passes(tr) {
    if (state.st === 'overdue') { if (!isOverdue(tr)) return false; }
    else if (state.st && tr.dataset.status !== state.st) return false;
    if (state.view === 'open' && !(tr.dataset.status === 'sent' || tr.dataset.status === 'partial')) return false;
    if (state.view === 'draft' && tr.dataset.status !== 'draft') return false;
    if (state.view === 'paid' && tr.dataset.status !== 'paid') return false;
    if (state.view === 'overdue' && !isOverdue(tr)) return false;
    if (period.from && tr.dataset.date < period.from) return false;
    if (period.to && tr.dataset.date > period.to) return false;
    if (state.q && tr.dataset.search.toLowerCase().indexOf(state.q) === -1) return false;
    return true;
  }
  function sortRows(list) {
    var k = state.sort;
    return list.sort(function (a, b) {
      if (k === 'new') return a.dataset.date < b.dataset.date ? 1 : -1;
      if (k === 'old') return a.dataset.date > b.dataset.date ? 1 : -1;
      if (k === 'high') return Number(b.dataset.total) - Number(a.dataset.total);
      if (k === 'bal') return (balOf(IV[b.dataset.id]) - balOf(IV[a.dataset.id]));
      if (k === 'due') return a.dataset.due < b.dataset.due ? -1 : 1;
      return 0;
    });
  }

  function presetRange(v) {
    var y = TODAY.getFullYear(), m = TODAY.getMonth();
    if (v === 'm0') return [localISO(y, m, 1), localISO(y, m + 1, 0)];
    if (v === 'm-1') return [localISO(y, m - 1, 1), localISO(y, m, 0)];
    if (v === 'd30') return [addDaysISO(TODAY_ISO, -29), TODAY_ISO];
    if (v === 'q0') { var qs = Math.floor(m / 3) * 3; return [localISO(y, qs, 1), localISO(y, qs + 3, 0)]; }
    if (v === 'ytd') return [y + '-01-01', localISO(y, 11, 31)];
    return null;
  }

  function updateSplitSum() {

  $('pySave').addEventListener('click', function () {
    var iv = IV[payId], tr = rowOf(payId), ref = tr.querySelector('.ino').textContent;
    var bal = balOf(iv);
    var amt = num($('pyAmount').value);
    if (!(amt > 0)) { $('pyAmount').classList.add('bad'); toast('Enter a payment amount greater than zero'); return; }
    if (amt > bal) { $('pyAmount').classList.add('bad'); toast('Payment cannot exceed the open balance (K ' + fmt(bal) + ')'); return; }
    $('pyAmount').classList.remove('bad');
    var m = $('pyMethod').value;
    var details = {}, splits = null, methodLabel = m;
    if (m === 'Split') {
      splits = splitLegs().filter(function (l) { return l.amount > 0 || l.ref; });
      if (splits.length < 2) { toast('A split payment needs at least two legs with amounts'); return; }
      var sum = splits.reduce(function (s, l) { return s + l.amount; }, 0);
      if (Math.abs(sum - amt) > 0.005) { toast('Split legs (K ' + fmt(sum) + ') must equal the payment amount (K ' + fmt(amt) + ')'); return; }
      for (var i = 0; i < splits.length; i++) if (!(splits[i].amount > 0)) { toast('Every split leg needs an amount greater than zero'); return; }
      methodLabel = 'Split · ' + splits.map(function (l) { return l.method; }).join(' + ');
    } else {
      details = collectDetails(m);
      var missing = [];
      (PYFIELDS[m] || []).forEach(function (f) { if (f.req && !details[f.k]) missing.push(f.l.toLowerCase()); });
      if (missing.length) { toast('Complete: ' + missing.join(', ')); return; }
    }
    var p = { date: $('pyDate').value || TODAY_ISO, method: m, methodLabel: methodLabel, ref: $('pyReceipt').value, amount: amt, note: $('pyNote').value.trim(), details: details, splits: splits };
    var settled = Number((bal - amt).toFixed(2)) <= 0;
    var rowsHtml = cbRow('Receipt №', p.ref, 'mono') + cbRow('Payment date', dfmt(p.date)) + cbRow('Method', methodLabel) + cbRow('Payment amount', 'K ' + fmt(amt), 'green') + cbRow('Balance before', 'K ' + fmt(bal)) + cbRow('Balance after', 'K ' + fmt(bal - amt), settled ? 'green' : '');
    if (splits) {
      rowsHtml += splits.map(function (l) { return cbRow('Leg — ' + l.method, 'K ' + fmt(l.amount) + (l.ref ? ' · ' + l.ref : ''), 'mono'); }).join('');
    } else {
      (PYFIELDS[m] || []).forEach(function (f) { if (details[f.k]) rowsHtml += cbRow(f.l, details[f.k], f.k === 'txref' || f.k === 'chq' ? 'mono' : ''); });
    }
    if (p.note) rowsHtml += cbRow('Note', p.note);
    ask({
      tone: 'ok',
      title: 'Confirm payment receipt',
      okLabel: 'Confirm payment',
      msg: 'Post this payment against <span class="okref">' + ref + '</span> (' + iv.cust + ')?<div class="cb-box">' + rowsHtml + '</div>' +
        (settled ? '<div style="margin-top:10px;font-weight:700;color:var(--green)">This payment settles the invoice in full.</div>' : '<div style="margin-top:10px;font-weight:600;color:var(--amber-2)">Partial payment — invoice moves to Partially paid.</div>'),
      onOk: function () { applyPayment(payId, p); }
    });
  });
