    PUT /receipts/{id} (409 unless draft) · POST /receipts/{id}/post ·
    POST /receipts/{id}/void (409 when posted allocations already reversed twice) ·
    POST /receipts/{id}/email (queues job; 422 invalid email) ·
    POST /receipts/{id}/email/retry · GET /receipts/email-jobs/{job} (stage/status/reason)
    or SSE stream · GET /receipts/{id}/print?format=html|pdf ·
    GET /reports/receipts/daily-summary | cashbook | method-mix · POST /receipts/export.
    Posting applies allocations to invoice balances in ONE transaction (invoices module
    hook); voiding reverses them; on-account remainder posts to a customer-advance
    ledger view consumed by the invoices module's allocation editor.

    emits the four stage events of R19; failures store SMTP code+reason in fail_reason
    and email_status='failed'; successes store emailed_to/at + 'sent'. Audit log rows
    for post, void, allocate, email-sent, email-failed, conversion-from-document.

    ref, customer, amount (balance or total), label, date; conversion writes source_type
    + source_ref on the receipt (quotation/order conversions flag the source doc as
    deposit-received where the module supports it).

  function passes(tr) {
    var rc = RC[tr.dataset.id];
    if (state.st === 'onacc') { if (!isOnAcc(rc)) return false; }
    else if (state.st && tr.dataset.status !== state.st) return false;
    if (state.view === 'draft' && tr.dataset.status !== 'draft') return false;
    if (state.view === 'posted' && tr.dataset.status !== 'posted') return false;
    if (state.view === 'onacc' && !isOnAcc(rc)) return false;
    if (state.view === 'void' && tr.dataset.status !== 'void') return false;
    if (period.from && tr.dataset.date < period.from) return false;
    if (period.to && tr.dataset.date > period.to) return false;
    if (state.q && tr.dataset.search.toLowerCase().indexOf(state.q) === -1) return false;
    return true;
  }
  function sortRows(list) {
    var k = state.sort;
    return list.sort(function (a, b) {
      if (k === 'new') return a.dataset.date < b.dataset.date ? 1 : -1;
      if (k === 'old') return a.dataset.date > b.dataset.date ? 1 : -1;
      if (k === 'high') return Number(b.dataset.total) - Number(a.dataset.total);
      if (k === 'low') return Number(a.dataset.total) - Number(b.dataset.total);
      return 0;
    });
  }

  function paintProgress() {
    var scrim = $('mailScrim');
    var fill = scrim.querySelector('#mpFill'), pctEl = scrim.querySelector('#mpPct'), logEl = scrim.querySelector('#mpLog');
    if (fill) fill.style.width = mail.pct + '%';
    if (pctEl) pctEl.textContent = Math.round(mail.pct) + '%';
    if (logEl) logEl.textContent = mail.status;
    var dock = $('mailDock');
    if (dock && !mail.done) {
      var df = dock.querySelector('#dkFill'), dp = dock.querySelector('#dkPct'), dst = dock.querySelector('#dkStatus');
      if (df) df.style.width = mail.pct + '%';

    }
  }
  function runPipeline(rc) {
    var scrim = $('mailScrim');
    var steps = scrim.querySelectorAll('.mp-step');
    function act(i) { steps[i].classList.add('active'); }
    function done(i) { steps[i].classList.remove('active'); steps[i].classList.add('done'); }
    function fail(i) { steps[i].classList.remove('active'); steps[i].classList.add('fail'); }
    function at(t, fn) { mail.timers.push(setTimeout(fn, t)); }
    mail.interval = setInterval(function () {
      if (mail.pct < mail.target) {
        mail.pct = Math.min(mail.target, mail.pct + 1.1);
        paintProgress();
      }
    }, 45);
    var willFail = !!rc.simFail;
    /* stage 1 · preparing pdf document */
    mail.target = 22; act(0); mail.status = 'Preparing PDF document…'; paintProgress();
    /* stage 2 · uploading to mail server */
    at(1100, function () { done(0); mail.target = 48; act(1); mail.status = 'Uploading to mail server…'; paintProgress(); });
    /* stage 3 · delivering to recipient mailbox */
    at(2300, function () { done(1); mail.target = 78; act(2); mail.status = 'Delivering to recipient mailbox…'; paintProgress(); });
    at(3300, function () {
      if (willFail) { fail(2); finishMail(false, '552 5.2.2 Mailbox full — message rejected', rc); return; }
      mail.status = 'Delivering to recipient mailbox…'; paintProgress();
    });
    /* stage 4 · recording email dispatch audit */
    at(4100, function () { if (mail.done) return; done(2); mail.target = 96; act(3); mail.status = 'Recording email dispatch audit…'; paintProgress(); });
    at(4900, function () { if (mail.done) return; done(3); mail.target = 100; finishMail(true, '', rc); });
  }
  function finishMail(ok, reason, rc) {
    mailClear();
    mail.done = true; mail.ok = ok; mail.running = true;
    if (ok) mail.pct = 100;
    var scrim = $('mailScrim');
    if (ok) {
      mail.status = 'Done — official receipt delivered.';
      scrim.classList.add('done');
      scrim.querySelector('#mpTitle').textContent = 'Receipt delivered';
      scrim.querySelector('#mpSub').textContent = 'Accepted by ' + (String(rc.email).split('@')[1] || 'recipient') + ' · dispatch recorded';
      rc.emailStatus = 'sent'; rc.failReason = ''; rc.emailed = rc.email; rc.emailedAt = TODAY_ISO;
      delete rc.simFail;
      setMpFoot('success');
    } else {
      mail.status = 'Delivery failed — ' + reason;
      scrim.classList.add('failed');
      scrim.querySelector('#mpTitle').textContent = 'Delivery failed';
      scrim.querySelector('#mpSub').textContent = reason + ' · ' + rc.email;
      rc.emailStatus = 'failed'; rc.failReason = reason;
      setMpFoot('failed');
    }
    $('mpMin').style.display = 'none';
    paintProgress();
    replaceRow(mail.id, rowPayload(mail.id, rc.status));
    refresh();
    if (mail.minimized) {
      dockResult(ok, rc, reason);
      toast(ok ? '📨 ' + mail.ref + ' delivered to ' + rc.email : '⚠ ' + mail.ref + ' delivery failed — ' + reason);
    }
  }

  function validateReceipt(post) {
    var walkin = $('crMode').value === 'walkin';
    var name = walkin ? $('crName').value.trim() : $('crSearch').value.trim();
    if (!name) { (walkin ? $('crName') : $('crSearch')).classList.add('bad'); toast(walkin ? 'Payer name is required' : 'Select a customer from the registry'); return null; }
    (walkin ? $('crName') : $('crSearch')).classList.remove('bad');
    if (!walkin && !matchedCode) { $('crSearch').classList.add('bad'); toast('Pick a matching customer from the list'); return null; }
    $('crSearch').classList.remove('bad');
    var amt = num($('crAmount').value);
    if (!(amt > 0)) { $('crAmount').classList.add('bad'); toast('Enter an amount received greater than zero'); return null; }
    $('crAmount').classList.remove('bad');
    if (!$('crDate').value) { $('crDate').classList.add('bad'); toast('Receipt date is required'); return null; }
    $('crDate').classList.remove('bad');
    var miss = crDetailsMissing();
    if (miss.length) { toast('Complete: ' + miss.join(', ')); return null; }
    var allocs = allocRows();
    var al = allocs.reduce(function (s, a) { return s + a.amount; }, 0);
    if (al > amt + 0.005) { toast('Allocations (K ' + fmt(al) + ') cannot exceed the amount received (K ' + fmt(amt) + ')'); return null; }
    return {
      walkin: walkin, name: name,
      contact: $('crContact').value.trim(), email: $('crEmail').value.trim(), phone: $('crPhone').value.trim(),
      date: $('crDate').value, method: $('crMethod').value, details: collectCrDetails(),
      amount: amt, allocs: allocs, by: $('crBy').value.trim() || 'Elvis Seyama',
      source: $('crSource').value.trim(), notes: $('crNotes').value.trim(), post: post
    };
  }
